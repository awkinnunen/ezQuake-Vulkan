cmake_minimum_required(VERSION 3.24)
project(ezv_friends_probe LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This first probe uses Windows WinHTTP for the existing FTE broker")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3.2 REQUIRED)
find_package(LibJuice CONFIG REQUIRED)
add_executable(FriendsProbe main.cpp broker.cpp protocol.cpp)
target_compile_definitions(FriendsProbe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0602)
target_compile_options(FriendsProbe PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(FriendsProbe PRIVATE LibJuice::LibJuice OpenSSL::SSL OpenSSL::Crypto winhttp ws2_32 crypt32)
enable_testing()
add_test(NAME friends_protocol COMMAND FriendsProbe --self-test)
